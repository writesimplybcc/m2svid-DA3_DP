# StereoFaster - Requirements
#
# This file is intended for:
# - Local development (pip install -r requirements.txt)
# - Git tracking of dependencies
# - Docker / CI base layer
#
# Notes:
# - PyTorch + CUDA **must** be installed first (before this file).
# - Recommended versions (as of 2026):
#     • PyTorch 2.3.1 + CUDA 12.1   → Best balance for both DA3 and M2SVid/sgm
#     • PyTorch 2.4.x + CUDA 12.1   → Also works well (newer)
#     • CUDA 11.8 users: Use PyTorch 2.0.1 + cu118 (matches original M2SVid env)
#
# Install PyTorch first with (CUDA 12.1 example):
#   pip install torch==2.3.1 torchvision==0.18.1 torchaudio==2.3.1 --index-url https://download.pytorch.org/whl/cu121
#
# Then run:
#   pip install -r requirements.txt
#
# xformers is highly recommended by DA3 but can be skipped on some platforms.
# This covers the modern 2-step DA3 + M2SVid pipeline + the Gradio WebUI.

# Docker Example (recommended pattern)

# FROM nvidia/cuda:12.1.1-cudnn8-devel-ubuntu22.04

# RUN pip install --upgrade pip
# RUN pip install torch==2.3.1 torchvision==0.18.1 torchaudio==2.3.1 \
#     --index-url https://download.pytorch.org/whl/cu121
# RUN pip install -r requirements.txt
#
# Then install DA3 in editable mode inside the container:
#   cd Depth-Anything-3 && pip install -e ".[app]"
#
# And make sure third-party paths are in PYTHONPATH for M2SVid:
